  45      data.table::setcolorder(Curve, c("ky", "p", "Dn", "IDn", "w"))

==== newmark:::.buildKmaxDraws ====
   1  function (draws, Da) 
   2  {
   3      draws <- data.table::as.data.table(draws)
   4      Required <- c("ky", "s", "Dn", "IDn")
   5      Missing <- setdiff(Required, names(draws))
   6      if (length(Missing)) {
   7          stop("kmax draws are missing columns: ", paste(Missing, collapse = ", "))
   8      }
   9      if (!is.numeric(Da) || !length(Da) || anyNA(Da) || any(!is.finite(Da)) || any(Da <= 0) || anyDuplicated(Da)) {
  10          stop("Da must contain unique, finite positive values")
  11      }
  12      HasPGA <- "PGA.s" %in% names(draws)
  13      COLS <- c("ky", "s", "Dn", "IDn", if (HasPGA) "PGA.s")
  14      DT <- draws[, COLS, with = FALSE]
  15      DT[, `:=`(IDn, as.character(IDn))]
  16      if (!nrow(DT)) 
  17          stop("kmax draws are empty")
  18      if (anyNA(DT) || any(!is.finite(DT$ky)) || any(DT$ky <= 0) || any(!is.finite(DT$Dn)) || any(DT$Dn < 0) || any(!nzchar(DT$IDn))) {
  19          stop("kmax draws contain invalid ky, s, Dn, or IDn values")
  20      }
  21      if (HasPGA && (any(!is.finite(DT$PGA.s)) || any(DT$PGA.s <= 0))) {
  22          stop("kmax draws contain invalid PGA.s values")
  23      }
  24      if (anyDuplicated(DT, by = c("IDn", "s", "ky"))) {
  25          stop("kmax draws duplicate an IDn, s, ky key")
  26      }
  27      OUT <- DT[order(ky), {
  28          if (.N < 2L) {
  29              stop("kmax inversion needs at least two ky values for IDn=", IDn[[1L]], " s=", s[[1L]])
  30          }
  31          Roots <- NULL
  32          Beyond <- NULL
  33          if (data.table::uniqueN(Dn) == 1L && Dn[[1L]] > 0 && identical(IDn[[1L]], "AM88")) {
  34              if (!HasPGA) {
  35                  stop("kmax inversion has non-invertible Dn support for IDn=AM88 s=", s[[1L]])
  36              }
  37              PGA <- unique(.SD[["PGA.s"]])
  38              if (length(PGA) != 1L) {
  39                  stop("flat AM88 inversion requires one PGA.s for IDn=AM88 s=", s[[1L]])
  40              }
  41              if (min(ky)/PGA < 0.9999) {
  42                  stop("kmax inversion has non-invertible Dn support for IDn=AM88 s=", s[[1L]])
  43              }
  44              Roots <- .invertFlatAM88(pga = PGA, dn = Dn[[1L]], Da = Da)
  45              Beyond <- rep(TRUE, length(Da))
  46          }
  47          if (is.null(Roots)) {
  48              x <- data.table::fifelse(Dn < FLOOR.Dn, 0, Dn)
  49              y <- rev(cummax(rev(x)))
  50              Roots <- numeric(length(Da))
  51              Beyond <- logical(length(Da))
  52              MASK <- Da <= y[[1L]]
  53              if (any(MASK)) {
  54                  if (data.table::uniqueN(y) < 2L) {
  55                    stop("kmax inversion has non-invertible Dn support for IDn=", IDn[[1L]], " s=", s[[1L]])
  56                  }
  57                  AUX <- .invertCurve(ky = ky, dn = x, Da = Da[MASK], idn = IDn[[1L]], pga = if (HasPGA) 
  58                    .SD[["PGA.s"]][[1L]]
  59                  else NULL)
  60                  Roots[MASK] <- AUX$root
  61                  Beyond[MASK] <- AUX$beyond
  62              }
  63          }
  64          .(kmax = list(Roots), beyond = list(Beyond))
  65      }, by = .(IDn, s)]
  66      OUT <- OUT[, .(Da = Da, kmax = unlist(kmax, use.names = FALSE), beyond = unlist(beyond, use.names = FALSE)), 
  67          by = .(IDn, s)]
  68      data.table::setcolorder(OUT, c("Da", "s", "kmax", "IDn", "beyond"))
  69      if (any(!is.finite(OUT$kmax)) || any(OUT$kmax < 0) || anyNA(OUT$beyond)) {
  70          stop("kmax inversion produced invalid roots")
  71      }
  72      OUT
  73  }

==== newmark:::.aggregateKmaxDraws ====
   1  function (draws, weights, p) 
   2  {
   3      if (!is.numeric(p) || anyNA(p) || any(!is.finite(p)) || any(p <= 0 | p >= 1) || anyDuplicated(p)) {
   4          stop("p must contain unique finite values in (0, 1)")
   5      }
   6      Labels <- sprintf("%.2f", p)
   7      if (anyDuplicated(Labels)) {
   8          stop("p values collide after two-decimal formatting")
   9      }
  10      if (!is.numeric(weights) || !length(weights) || is.null(names(weights)) || anyNA(names(weights)) || any(!nzchar(names(weights))) || 
  11          anyDuplicated(names(weights)) || anyNA(weights) || any(!is.finite(weights)) || any(weights < 0) || !any(weights > 
  12          0)) {
  13          stop("weights must be finite, non-negative, uniquely named, ", "and include one positive value")
  14      }
  15      Weights <- weights[weights > 0]
  16      Weights <- Weights/sum(Weights)
  17      Missing <- setdiff(names(Weights), unique(draws$IDn))
  18      if (length(Missing)) {
  19          stop("kmax draws are missing positive-weight models: ", paste(Missing, collapse = ", "))
  20      }
  21      DT <- draws[IDn %in% names(Weights)]
  22      Models <- sort(names(Weights))
  23      Weights <- Weights[Models]
  24      Samples <- lapply(Models, function(model) sort(DT[IDn == model, unique(s)]))
  25      Reference <- Samples[[1L]]
  26      OK <- vapply(Samples, identical, logical(1), y = Reference)
  27      if (!all(OK)) {
  28          stop("kmax draws require complete shared s coverage")
  29      }
  30      DT[, {
  31          LIST <- lapply(Models, function(model) kmax[IDn == model])
  32          AUX <- .aggregateWeighted(LIST, as.numeric(Weights), p)
  33          Share <- sum(vapply(seq_along(Models), function(i) Weights[[i]] * mean(beyond[IDn == Models[[i]]]), numeric(1)))
  34          data.table::data.table(p = c("mean", Labels), kmax = c(AUX[p == "mean", Dn], AUX[p != "mean", Dn]), beyond = Share)
  35      }, by = "Da"]
  36  }

==== newmark:::.sampleDnScenario ====
   1  function (uhs, Ts, NS, zTrunc = NULL) 
   2  {
   3      uhs <- data.table::as.data.table(uhs)
   4      InputType <- .classifySaInput(uhs = uhs)
   5      IsAnalytic <- identical(InputType, "analytic")
   6      Periods <- c(0, 1.3 * Ts, 1.5 * Ts)
   7      if (IsAnalytic) {
   8          SaTable <- uhs
   9      }
  10      else {
  11          uhs <- uhs[, .(Tn, p, Sa)]
  12          uhs[, `:=`(p, as.character(p))]
  13          LIST <- lapply(Periods, function(period) {
  14              if (period %in% uhs$Tn) {
  15                  uhs[Tn == period]
  16              }
  17              else {
  18                  interpolateSaTable(uhs = uhs, Tn = period)
  19              }
  20          })
  21          SaTable <- data.table::rbindlist(LIST, use.names = TRUE)
  22      }
  23      J <- length(Periods)
  24      C <- diag(J)
  25      for (j in seq_len(J)) for (k in seq_len(J)) {
  26          if (j != k) {
  27              Tj <- if (Periods[j] == 0) 
  28                  0.01
  29              else Periods[j]
  30              Tk <- if (Periods[k] == 0) 
  31                  0.01
  32              else Periods[k]
  33              C[j, k] <- rhoBJ(Tn = Tj, T0 = Tk)
  34          }
  35      }
  36      SaMatrix <- .sampleSaCorrFull(uhs = SaTable, Tn = Periods, rho = C, NS = NS, zTrunc = zTrunc)
  37      PGA <- SaMatrix[, 1L]
  38      Sa13 <- SaMatrix[, 2L]
  39      Sa15 <- SaMatrix[, 3L]
  40      PGV <- (PGA^1.0529) * exp(0.1241) * 100
  41      AI <- (PGA^1.9228) * exp(2.6109)
  42      if (IsAnalytic) {
  43          PValues <- c(0.05, 0.1, 0.16, 0.5, 0.84, 0.9, 0.95)
  44      }
  45      else {
  46          x <- suppressWarnings(as.numeric(uhs[p != "mean", p]))
  47          PValues <- sort(unique(x[is.finite(x)]))
  48      }
  49      list(PGA = PGA, Sa13 = Sa13, Sa15 = Sa15, PGV = PGV, AI = AI, pVals = PValues)
  50  }
